## API Contracts (from Core API)

### Search Request (GET /api/v1/search)
```typescript
interface SearchRequest {
  q: string;                    // Search query (required)
  domains: string[];            // Domain IDs to search (comma-separated)
  mode?: 'semantic' | 'keyword' | 'hybrid';  // Search mode (default: hybrid)
  top_k?: number;               // Number of results (default: 10)
  type?: string;                // Filter by document type (pdf, txt, etc.)
  date_from?: string;           // ISO date filter from
  date_to?: string;             // ISO date filter to
  source?: string;              // Filter by source type
}
```

### Search Response
```typescript
interface SearchResponse {
  query: string;
  results: SearchResult[];
  total: number;
  search_time_ms: number;
}

interface SearchResult {
  chunk_id: string;
  score: number;                // Relevance score 0-1
  text: string;                 // Chunk text content
  document_id: string;
  document_title: string;
  domain_id: string;
  metadata: {
    type?: string;
    source?: string;
    created_at?: string;
    // ... other metadata
  };
}
```

### Domain List Response (GET /api/v1/domains)
```typescript
interface DomainListResponse {
  items: Domain[];
  total: number;
  page: number;
  page_size: number;
  pages: number;
}

interface Domain {
  id: string;
  name: string;
  description?: string;
  document_count: number;
  // ... other fields
}
```
