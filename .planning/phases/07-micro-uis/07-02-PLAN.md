## API Contracts (from Core API)

### Domain List (GET /api/v1/domains)
```typescript
interface DomainListResponse {
  items: Domain[];
  total: number;
  page: number;
  page_size: number;
  pages: number;
}

interface Domain {
  id: string;
  name: string;
  description?: string;
  embedding_model: string;
  embedding_dimension: number;
  document_count: number;       // Key field for display
  created_by: string;
  created_at: string;
  updated_at: string;
}
```

### Domain Detail (GET /api/v1/domains/{id})
```typescript
interface DomainDetail extends Domain {
  // Same as Domain
}
```

### Documents in Domain (GET /api/v1/domains/{id}/documents - from domain_service)
```typescript
interface DocumentListResponse {
  items: Document[];
  total: number;
  page: number;
  page_size: number;
  pages: number;
}

interface Document {
  id: string;
  title: string;
  domain_id: string;
  source_type: string;          // upload, s3, kafka, api
  source_uri?: string;
  status: 'pending' | 'processing' | 'done' | 'failed';
  chunk_count: number;
  error_message?: string;
  created_at: string;
  updated_at: string;
  metadata: {
    content_type?: string;
    original_filename?: string;
    size_bytes?: number;
    // ... other metadata
  };
}
```

Note: The documents endpoint may need to be added to the API if not present. Check api/src/api/domains.py for /{domain_id}/documents endpoint.
