    Script:
    ```typescript
    import { ref, shallowRef, onMounted, watch } from 'vue'
    
    const props = defineProps<{
      remoteName: string
      componentName?: string
    }>()
    
    const loading = ref(true)
    const error = ref<string | null>(null)
    const loadedComponent = shallowRef(null)
    
    async function loadRemote() {
      loading.value = true
      error.value = null
      
      try {
        // @ts-ignore - Module Federation dynamic import
        const remote = await import(/* @vite-ignore */ `${props.remoteName}/${props.componentName || 'App'}`)
        loadedComponent.value = remote.default || remote
      } catch (err) {
        console.error(`Failed to load remote ${props.remoteName}:`, err)
        error.value = err instanceof Error ? err.message : 'Unknown error'
      } finally {
        loading.value = false
      }
    }
    
    function retry() {
      loadRemote()
    }
    
    onMounted(loadRemote)
    watch(() => props.remoteName, loadRemote)
    ```

    Create `frontend/apps/shell/src/composables/useRemote.ts`:

    ```typescript
    import { ref, shallowRef, onMounted, watch, type Component } from 'vue'

    interface UseRemoteOptions {
      remoteName: string
      componentName?: string
      immediate?: boolean
    }

    interface UseRemoteReturn {
      component: ReturnType<typeof shallowRef<Component | null>>
      loading: ReturnType<typeof ref<boolean>>
      error: ReturnType<typeof ref<string | null>>
      load: () => Promise<void>
      retry: () => Promise<void>
    }

    export function useRemote(options: UseRemoteOptions): UseRemoteReturn {
      const { remoteName, componentName = 'App', immediate = true } = options
      
      const component = shallowRef<Component | null>(null)
      const loading = ref(false)
      const error = ref<string | null>(null)

      async function load() {
        loading.value = true
        error.value = null
        component.value = null

        try {
          // @ts-ignore - Module Federation dynamic import
          const remote = await import(/* @vite-ignore */ `${remoteName}/${componentName}`)
          component.value = remote.default || remote
        } catch (err) {
          console.error(`Failed to load remote ${remoteName}:`, err)
          error.value = err instanceof Error 
            ? err.message 
            : `Failed to load ${remoteName}`
        } finally {
          loading.value = false
        }
      }

      async function retry() {
        await load()
      }

      if (immediate) {
        onMounted(load)
        watch(() => remoteName, load)
      }

      return {
        component,
        loading,
        error,
        load,
        retry
      }
    }

    // Helper to check if a remote is available
    export async function checkRemoteAvailability(remoteName: string): Promise<boolean> {
      try {
        // @ts-ignore
        await import(/* @vite-ignore */ `${remoteName}/App`)
        return true
      } catch {
        return false
      }
    }

    // Helper to preload a remote
    export async function preloadRemote(remoteName: string): Promise<void> {
      try {
        // @ts-ignore
        await import(/* @vite-ignore */ `${remoteName}/App`)
      } catch (err) {
        console.warn(`Failed to preload ${remoteName}:`, err)
      }
    }
    ```
